import { useState } from 'react';
import Navbar from '@/components/layout/Navbar';
import Footer from '@/components/layout/Footer';
import Hero from '@/components/home/Hero';
import Services from '@/components/home/Services';
import Doctors from '@/components/home/Doctors';
import HowItWorks from '@/components/home/HowItWorks';
import BookingSection, { type DoctorPreselection } from '@/components/booking/BookingSection';

function App() {
  const directory = useDoctors();
  const [preselection, setPreselection] = useState<DoctorPreselection | null>(null);

    setPreselection({ doctorId });

export default App;

export const SLOT_TIMES = [

const SUPABASE_URL = import.meta.env.VITE_SUPABASE_URL;
const SUPABASE_ANON_KEY = import.meta.env.VITE_SUPABASE_ANON_KEY;

export function toDateInputValue(date: Date): string {
  const y = date.getFullYear();
  const m = String(date.getMonth() + 1).padStart(2, '0');
  const d = String(date.getDate()).padStart(2, '0');
  return `${y}-${m}-${d}`;
}

export function weekdayOf(date: string): number {
  const day = new Date(`${date}T00:00:00Z`).getUTCDay();
  return day === 0 ? 7 : day;
}

export function isSlotInPast(date: string, time: string, now: Date): boolean {
  return new Date(`${date}T${time}:00`).getTime() <= now.getTime();
}

export function formatDisplayDate(date: string): string {
  return new Date(`${date}T00:00:00`).toLocaleDateString('en-GB', {
    weekday: 'long',
    day: 'numeric',
    month: 'long',
    year: 'numeric',
  });
}

export function formatDisplayTime(time: string): string {
  const [h, m] = time.split(':').map(Number);
  const period = h >= 12 ? 'PM' : 'AM';
  const hour12 = h % 12 === 0 ? 12 : h % 12;
  return `${hour12}:${String(m).padStart(2, '0')} ${period}`;
}

export async function fetchBookedSlots(doctorId: string, date: string): Promise<string[]> {
  const { data, error } = await supabase.rpc('get_booked_slots', {
    p_doctor_id: doctorId,
    p_date: date,
  });
  if (error) throw new Error('Could not load available time slots.');
  if (!Array.isArray(data)) throw new Error('Could not load available time slots.');
  return data.filter((t): t is string => typeof t === 'string').map((t) => t.slice(0, 5));
}

export async function createAppointment(input: BookingInput): Promise<BookingResult> {
  const response = await fetch(`${SUPABASE_URL}/functions/v1/book-appointment`, {
    method: 'POST',
    headers: {
      Authorization: `Bearer ${SUPABASE_ANON_KEY}`,
      apikey: SUPABASE_ANON_KEY,
      'Content-Type': 'application/json',
    },
    body: JSON.stringify(input),
  });

  const body = await response.json().catch(() => null);
  if (!response.ok) {
    const message = body && typeof body.error === 'string' ? body.error : 'Booking failed. Please try again.';
    throw new Error(message);
  }
  if (!body || typeof body.appointmentId !== 'string' || !body.notifications) {
    throw new Error('Booking failed. Please try again.');
  }
  return body as BookingResult;
}

import { useCallback, useEffect, useState } from 'react';
import { supabase } from '@/lib/supabase';
import type { Doctor } from '@/lib/types';

type DoctorsState =
  | { status: 'loading'; doctors: [] }
  | { status: 'ready'; doctors: Doctor[] }
  | { status: 'error'; doctors: [] };

export type DoctorDirectory = ReturnType<typeof useDoctors>;

export function useDoctors() {
  const [state, setState] = useState<DoctorsState>({ status: 'loading', doctors: [] });

  const load = useCallback(async () => {
    setState({ status: 'loading', doctors: [] });
    const { data, error } = await supabase
      .from('doctors')
      .select('id, name, specialty, qualification, experience_years, bio, working_days')
      .order('name');
    if (error || !Array.isArray(data)) {
      setState({ status: 'error', doctors: [] });
      return;
    }
    setState({ status: 'ready', doctors: data as Doctor[] });
  }, []);

  useEffect(() => {
    load();
  }, [load]);

  return { ...state, reload: load };
}

import { useEffect, useState } from 'react';
import { ArrowRight, CalendarDays, UserRound } from 'lucide-react';
import type { Doctor, BookingResult } from '@/lib/types';
import { createAppointment } from '@/lib/booking';
import ScheduleStep, { type ScheduleSelection } from '@/components/booking/ScheduleStep';
import PatientDetailsForm, { type PatientDetails } from '@/components/booking/PatientDetailsForm';
import BookingConfirmation from '@/components/booking/BookingConfirmation';

export type DoctorPreselection = { doctorId: string };

type BookingSectionProps = {
  doctors: Doctor[];
  preselection: DoctorPreselection | null;
};

type Step = 'schedule' | 'details' | 'done';

const INDICATOR_STEPS = [
  { id: 'schedule', label: 'Doctor & time', icon: CalendarDays },
  { id: 'details', label: 'Your details', icon: UserRound },
] as const;

const EMPTY_DETAILS: PatientDetails = { name: '', email: '', phone: '', reason: '' };

export default function BookingSection({ doctors, preselection }: BookingSectionProps) {
  const [step, setStep] = useState<Step>('schedule');
  const [selection, setSelection] = useState<ScheduleSelection>({
    doctorId: preselection?.doctorId ?? '',
    date: '',
    time: '',
  });
  useEffect(() => {
    if (!preselection) return;
    setSelection((s) => ({ ...s, doctorId: preselection.doctorId, time: '' }));
    setStep('schedule');
  }, [preselection]);

  const [details, setDetails] = useState<PatientDetails>(EMPTY_DETAILS);
  const [submitting, setSubmitting] = useState(false);
  const [submitError, setSubmitError] = useState<string | null>(null);
  const [result, setResult] = useState<BookingResult | null>(null);
  const [bookedFor, setBookedFor] = useState<{ doctorName: string; date: string; time: string } | null>(null);

  const doctor = doctors.find((d) => d.id === selection.doctorId);
  const canContinue = !!doctor && !!selection.date && !!selection.time;
  const currentStepIndex = INDICATOR_STEPS.findIndex((s) => s.id === step);

  const handleSubmit = async () => {
    if (!doctor) return;
    setSubmitting(true);
    setSubmitError(null);
    try {
      const booking = await createAppointment({
        doctorId: selection.doctorId,
        date: selection.date,
        time: selection.time,
        patientName: details.name.trim(),
        patientEmail: details.email.trim(),
        patientPhone: details.phone.trim(),
        reason: details.reason.trim(),
      });
      setResult(booking);
      setBookedFor({ doctorName: doctor.name, date: selection.date, time: selection.time });
      setStep('done');
    } catch (err) {
      setSubmitError(err instanceof Error ? err.message : 'Booking failed. Please try again.');
    } finally {
      setSubmitting(false);
    }
  };

  const resetBooking = () => {
    setSelection({ doctorId: '', date: '', time: '' });
    setDetails(EMPTY_DETAILS);
    setResult(null);
    setBookedFor(null);
    setSubmitError(null);
    setStep('schedule');
  };

  return (
    <section id="book" className="scroll-mt-20 bg-gradient-to-b from-white to-primary-50/60 py-20 lg:py-28">
      <div className="mx-auto max-w-3xl px-6 lg:px-8">
        <div className="text-center">
          <p className="text-sm font-bold uppercase tracking-wider text-accent-600">Book online</p>
          <h2 className="mt-3 text-3xl font-bold tracking-tight text-primary-900 sm:text-4xl">
            Schedule your consultation
          </h2>
          <p className="mt-4 text-lg leading-relaxed text-slate-600">
            Pick a doctor, a date and a time slot. Both you and your doctor receive confirmation by email and SMS.
          </p>
        </div>

        <div className="mt-10 overflow-hidden rounded-3xl bg-white shadow-xl shadow-primary-900/5 ring-1 ring-slate-100">
          {step !== 'done' && (
            <ol className="grid grid-cols-2 border-b border-slate-100 bg-slate-50/70">
              {INDICATOR_STEPS.map((s, index) => {
                const active = index === currentStepIndex;
                const complete = index < currentStepIndex;
                const Icon = s.icon;
                return (
                  <li
                    key={s.id}
                    className={`flex items-center justify-center gap-2 px-3 py-4 text-sm font-semibold transition-colors ${
                      active ? 'bg-white text-primary-700' : complete ? 'text-accent-700' : 'text-slate-400'

                  >
                    <span
                      className={`flex h-7 w-7 items-center justify-center rounded-full text-xs ${
                        active
                          ? 'bg-primary-600 text-white'
                          : complete
                            ? 'bg-accent-500 text-white'
                            : 'bg-slate-200 text-slate-500'
                      }`}
                    >
                      {index + 1}
                    </span>
                    <span className="hidden sm:inline">{s.label}</span>
                    <Icon className="h-4 w-4 sm:hidden" />
                  </li>
                );
              })}
            </ol>
          )}

          <div className="p-6 sm:p-10">
            {step === 'schedule' && (
              <div className="animate-fade-up">
                {doctors.length === 0 ? (
                  <p className="text-center text-slate-600">Doctors are not available for booking right now.</p>
                ) : (
                  <ScheduleStep doctors={doctors} selection={selection} onChange={setSelection} />
                )}
                <div className="mt-10 flex justify-end">
                  <button
                    type="button"
                    disabled={!canContinue}
                    onClick={() => setStep('details')}
                    className="inline-flex items-center gap-2 rounded-full bg-primary-600 px-8 py-3 text-sm font-semibold text-white shadow-lg shadow-primary-600/25 transition-all hover:-translate-y-0.5 hover:bg-primary-700 disabled:cursor-not-allowed disabled:bg-slate-300 disabled:shadow-none disabled:hover:translate-y-0"
                  >
                    Continue
                    <ArrowRight className="h-4 w-4" />

                </div>

import { useEffect, useState } from 'react';
import { AlertCircle, Loader2 } from 'lucide-react';
import type { Doctor } from '@/lib/types';
import {
  fetchBookedSlots,
  toDateInputValue,
  weekdayOf,
  formatDisplayDate,
} from '@/lib/booking';
import SlotPicker from '@/components/booking/SlotPicker';

const BOOKING_WINDOW_DAYS = 60;

export type ScheduleSelection = {
  doctorId: string;
  date: string;
  time: string;
};

type ScheduleStepProps = {
  doctors: Doctor[];
  selection: ScheduleSelection;
  onChange: (next: ScheduleSelection) => void;
};

export default function ScheduleStep({ doctors, selection, onChange }: ScheduleStepProps) {
  const [now] = useState(() => new Date());
  const [bookedSlots, setBookedSlots] = useState<string[]>([]);
  const [slotsStatus, setSlotsStatus] = useState<'idle' | 'loading' | 'ready' | 'error'>('idle');
  const [reloadKey, setReloadKey] = useState(0);

  const { doctorId, date, time } = selection;
  const doctor = doctors.find((d) => d.id === doctorId);
  const dayAvailable = !!doctor && !!date && doctor.working_days.includes(weekdayOf(date));
  const showSlots = dayAvailable && !!doctor;

  useEffect(() => {
    if (!doctorId || !date || !dayAvailable) {
      setBookedSlots([]);
      setSlotsStatus('idle');
      return;
    }
    let cancelled = false;
    setSlotsStatus('loading');
    fetchBookedSlots(doctorId, date)
      .then((slots) => {
        if (cancelled) return;
        setBookedSlots(slots);
        setSlotsStatus('ready');
      })
      .catch(() => {
        if (cancelled) return;
        setBookedSlots([]);
        setSlotsStatus('error');
      });
    return () => {
      cancelled = true;
    };
  }, [doctorId, date, dayAvailable, reloadKey]);

  const maxDate = new Date(now);
  maxDate.setDate(maxDate.getDate() + BOOKING_WINDOW_DAYS);

  return (
    <div className="space-y-8">
      <div>
        <label htmlFor="doctor" className="block text-sm font-semibold text-slate-700">
          Doctor
        </label>
        <select
          id="doctor"
          value={doctorId}
          onChange={(e) => onChange({ doctorId: e.target.value, date, time: '' })}
          className="mt-2 w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-slate-800 shadow-sm transition focus:border-primary-500 focus:outline-none focus:ring-4 focus:ring-primary-100"
        >
          <option value="">Choose a doctor</option>
          {doctors.map((d) => (
            <option key={d.id} value={d.id}>
              {d.name} - {d.specialty}
            </option>
          ))}
        </select>
      </div>

      <div>
        <label htmlFor="date" className="block text-sm font-semibold text-slate-700">
          Appointment date
        </label>
        <input
          id="date"
          type="date"
          min={toDateInputValue(now)}
          max={toDateInputValue(maxDate)}
          value={date}
          onChange={(e) => onChange({ doctorId, date: e.target.value, time: '' })}
          className="mt-2 w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-slate-800 shadow-sm transition focus:border-primary-500 focus:outline-none focus:ring-4 focus:ring-primary-100"
        />
        {doctor && date && !dayAvailable && (
          <p className="mt-2 text-sm text-amber-700">
            {doctor.name} does not see patients on {formatDisplayDate(date)}. Please choose another day.
          </p>
        )}
      </div>

      {showSlots && (
        <div>
          {slotsStatus === 'loading' && (
            <p className="flex items-center gap-2 text-sm text-slate-600">
              <Loader2 className="h-4 w-4 animate-spin text-primary-600" />
              Checking available slots...
            </p>
          )}
          {slotsStatus === 'error' && (
            <div className="flex items-center justify-between gap-4 rounded-xl border border-red-100 bg-red-50 p-4">
              <p className="flex items-center gap-2 text-sm text-red-700">
                <AlertCircle className="h-4 w-4" />
                We could not load the time slots.
              </p>
              <button
                type="button"
                onClick={() => setReloadKey((k) => k + 1)}
                className="text-sm font-semibold text-red-700 underline"
              >
                Retry
              </button>
            </div>
          )}
          {slotsStatus === 'ready' && (
            <SlotPicker
              date={date}
              bookedSlots={bookedSlots}
              selectedTime={time}
              now={now}
              onSelect={(t) => onChange({ doctorId, date, time: t })}
            />
          )}
        </div>
      )}
    </div>
  );
}

import { useState, type FormEvent, type ReactNode } from 'react';
import { Loader2 } from 'lucide-react';

export type PatientDetails = {
  name: string;
  email: string;
  phone: string;
  reason: string;
};

type PatientDetailsFormProps = {
  details: PatientDetails;
  onChange: (next: PatientDetails) => void;
  onBack: () => void;
  onSubmit: () => void;
  submitting: boolean;
  submitError: string | null;
};

const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const PHONE_RE = /^\+?[0-9\s()-]{7,20}$/;

const inputClass =
  'mt-2 w-full rounded-xl border border-slate-200 bg-white px-4 py-3 text-slate-800 shadow-sm transition placeholder:text-slate-400 focus:border-primary-500 focus:outline-none focus:ring-4 focus:ring-primary-100';

export default function PatientDetailsForm({
  details,
  onChange,
  onBack,
  onSubmit,
  submitting,
  submitError,
}: PatientDetailsFormProps) {
  const [errors, setErrors] = useState<Partial<Record<keyof PatientDetails, string>>>({});

  const handleSubmit = (e: FormEvent<HTMLFormElement>) => {
    e.preventDefault();
    const next: typeof errors = {};
    if (details.name.trim().length < 2) next.name = 'Please enter the patient name.';
    if (!EMAIL_RE.test(details.email.trim())) next.email = 'Please enter a valid email address.';
    if (!PHONE_RE.test(details.phone.trim())) next.phone = 'Please enter a valid phone number.';
    setErrors(next);
    if (Object.keys(next).length === 0) onSubmit();
  };

  const update = (field: keyof PatientDetails) => (value: string) => {
    onChange({ ...details, [field]: value });
  };

  return (
    <form onSubmit={handleSubmit} noValidate className="space-y-6">
      <div className="grid gap-6 sm:grid-cols-2">
        <Field label="Patient full name" error={errors.name}>
          <input
            type="text"
            autoComplete="name"
            value={details.name}
            onChange={(e) => update('name')(e.target.value)}
            className={inputClass}
            placeholder="e.g. Asha Verma"
          />
        </Field>
        <Field label="Phone number" error={errors.phone}>
          <input
            type="tel"
            autoComplete="tel"
            value={details.phone}
            onChange={(e) => update('phone')(e.target.value)}
            className={inputClass}
            placeholder="+91 98765 43210"
          />
        </Field>
      </div>

      <Field label="Email address" error={errors.email}>
        <input
          type="email"
          autoComplete="email"
          value={details.email}
          onChange={(e) => update('email')(e.target.value)}
          className={inputClass}
          placeholder="you@example.com"
        />
      </Field>

      <Field label="Reason for visit (optional)">
        <textarea
          rows={3}
          value={details.reason}
          onChange={(e) => update('reason')(e.target.value)}
          className={inputClass}
          placeholder="Briefly describe your symptoms or concern"
        />
      </Field>

      {submitError && (
        <p role="alert" className="rounded-xl border border-red-100 bg-red-50 px-4 py-3 text-sm text-red-700">
          {submitError}
        </p>
      )}

      <div className="flex flex-col-reverse gap-3 sm:flex-row sm:justify-between">
        <button
          type="button"
          onClick={onBack}
          disabled={submitting}
          className="rounded-full border border-slate-200 px-6 py-3 text-sm font-semibold text-slate-700 transition-colors hover:bg-slate-50 disabled:opacity-50"
        >
          Back
        </button>
        <button
          type="submit"
          disabled={submitting}

        >
          {submitting && <Loader2 className="h-4 w-4 animate-spin" />}
          {submitting ? 'Booking...' : 'Confirm appointment'}
        </button>
      </div>
    </form>
  );
}

function Field({ label, error, children }: { label: string; error?: string; children: ReactNode }) {

import { createClient } from "npm:@supabase/supabase-js@2";

const corsHeaders = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, PUT, DELETE, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization, X-Client-Info, Apikey",
};

const json = (body: unknown, status = 200) =>
  new Response(JSON.stringify(body), {
    status,
    headers: { ...corsHeaders, "Content-Type": "application/json" },
  });

type Booking = {
  doctorId: string;
  date: string;
  time: string;
  patientName: string;
  patientEmail: string;
  patientPhone: string;
  reason: string;
};

const UUID_RE = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i;
const DATE_RE = /^\d{4}-\d{2}-\d{2}$/;
const TIME_RE = /^([01]\d|2[0-3]):[0-5]\d$/;
const EMAIL_RE = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const PHONE_RE = /^\+?[0-9\s()-]{7,20}$/;

function parseBooking(input: unknown): Booking | string {
  if (!input || typeof input !== "object") return "Invalid request.";
  const b = input as Record<string, unknown>;
  const str = (v: unknown, max: number) =>
    typeof v === "string" ? v.trim().slice(0, max) : "";

  const booking: Booking = {
    doctorId: str(b.doctorId, 36),
    date: str(b.date, 10),
    time: str(b.time, 5),
    patientName: str(b.patientName, 100),
    patientEmail: str(b.patientEmail, 200),
    patientPhone: str(b.patientPhone, 20),
    reason: str(b.reason, 500),
  };

  if (!UUID_RE.test(booking.doctorId)) return "Please choose a doctor.";
  if (!DATE_RE.test(booking.date) || Number.isNaN(Date.parse(booking.date)))
    return "Please choose a valid date.";
  if (!TIME_RE.test(booking.time)) return "Please choose a valid time slot.";
  if (booking.patientName.length < 2) return "Please enter the patient's name.";
  if (!EMAIL_RE.test(booking.patientEmail)) return "Please enter a valid email address.";
  if (!PHONE_RE.test(booking.patientPhone)) return "Please enter a valid phone number.";
  return booking;
}

function isoWeekday(date: string): number {
  const day = new Date(`${date}T00:00:00Z`).getUTCDay();
  return day === 0 ? 7 : day;
}

async function sendEmail(to: string, subject: string, text: string): Promise<string> {
  const apiKey = Deno.env.get("RESEND_API_KEY");
  if (!apiKey) return "not_configured";
  const res = await fetch("https://api.resend.com/emails", {
    method: "POST",
    headers: {
      Authorization: `Bearer ${apiKey}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({
      from: Deno.env.get("RESEND_FROM") ?? "MHS <onboarding@resend.dev>",
      to: [to],
      subject,
      text,
    }),
  });
  return res.ok ? "sent" : "failed";
}

async function sendSms(to: string, body: string): Promise<string> {
  const sid = Deno.env.get("TWILIO_ACCOUNT_SID");
  const token = Deno.env.get("TWILIO_AUTH_TOKEN");
  const from = Deno.env.get("TWILIO_FROM_NUMBER");
  if (!sid || !token || !from) return "not_configured";
  const form = new URLSearchParams({ To: to, From: from, Body: body });
  const res = await fetch(`https://api.twilio.com/2010-04-01/Accounts/${sid}/Messages.json`, {
    method: "POST",
    headers: {
      Authorization: `Basic ${btoa(`${sid}:${token}`)}`,
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: form.toString(),
  });
  return res.ok ? "sent" : "failed";
}

Deno.serve(async (req: Request) => {
  if (req.method === "OPTIONS") {
    return new Response(null, { status: 200, headers: corsHeaders });
  }

  try {
    if (req.method !== "POST") return json({ error: "Method not allowed." }, 405);

    const parsed = parseBooking(await req.json().catch(() => null));
    if (typeof parsed === "string") return json({ error: parsed }, 400);
    const booking = parsed;

    const today = new Date().toISOString().slice(0, 10);
    if (booking.date < today) return json({ error: "Appointments must be in the future." }, 400);

    const supabase = createClient(
      Deno.env.get("SUPABASE_URL")!,
      Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!,
    );

    const { data: doctor, error: doctorError } = await supabase
      .from("doctors")
      .select("id, name, specialty, working_days, email, phone")
      .eq("id", booking.doctorId)
      .maybeSingle();
    if (doctorError) throw doctorError;
    if (!doctor) return json({ error: "That doctor could not be found." }, 404);

    if (!doctor.working_days.includes(isoWeekday(booking.date))) {
      return json({ error: `${doctor.name} is not available on that day.` }, 409);
    }

    const { data: appointment, error: insertError } = await supabase
      .from("appointments")
      .insert({
        doctor_id: booking.doctorId,
        appointment_date: booking.date,
        slot_time: booking.time,
        patient_name: booking.patientName,
        patient_email: booking.patientEmail,
        patient_phone: booking.patientPhone,
        reason: booking.reason,
      })
      .select("id")
      .single();

    if (insertError) {
      if (insertError.code === "23505") {
        return json({ error: "That time slot was just booked. Please pick another." }, 409);
      }
      throw insertError;
    }

    const when = `${booking.date} at ${booking.time}`;
    const patientText = `Hello ${booking.patientName},\n\nYour appointment with ${doctor.name} (${doctor.specialty}) at MHS is confirmed for ${when}.\n\nReference: ${appointment.id}\n\nThank you for choosing MHS.`;
    const doctorText = `Hello ${doctor.name},\n\nA new appointment has been booked with you for ${when}.\n\nPatient: ${booking.patientName}\nPhone: ${booking.patientPhone}\nEmail: ${booking.patientEmail}\nReason: ${booking.reason || "Not provided"}\n\nReference: ${appointment.id}`;
    const patientSms = `MHS: Your appointment with ${doctor.name} is confirmed for ${when}. Ref: ${appointment.id}`;
    const doctorSms = `MHS: New appointment with patient ${booking.patientName} on ${when}. Ref: ${appointment.id}`;

    const [patientEmailStatus, patientSmsStatus, doctorEmailStatus, doctorSmsStatus] =
      await Promise.all([
        sendEmail(booking.patientEmail, "Your MHS appointment is confirmed", patientText).catch(() => "failed"),
        sendSms(booking.patientPhone, patientSms).catch(() => "failed"),
        doctor.email
          ? sendEmail(doctor.email, "New MHS appointment booked", doctorText).catch(() => "failed")
          : Promise.resolve("no_contact"),
        doctor.phone
          ? sendSms(doctor.phone, doctorSms).catch(() => "failed")
          : Promise.resolve("no_contact"),
      ]);

    return json({
      appointmentId: appointment.id,
      notifications: {
        patientEmail: patientEmailStatus,
        patientSms: patientSmsStatus,
        doctorEmail: doctorEmailStatus,
        doctorSms: doctorSmsStatus,
      },
    });
  } catch (err) {
    console.error("book-appointment failed", err);
    return json({ error: "Something went wrong while booking. Please try again." }, 500);
  }
});

/*
# Create doctors and appointments for MHS medical booking

1. New Tables
- `doctors`: public directory of doctors (name, specialty, qualification, experience, bio, photo, weekly working days).
- `appointments`: booked consultations (doctor, date, time slot, patient name, email, phone, reason, status).

2. Security
- Enable RLS on both tables.
- `doctors`: readable by anon and authenticated (public directory). No writes from the browser.
- `appointments`: no direct browser access at all. Patient details stay private; bookings are created by a server function using the service role.
- Function `get_booked_slots`: returns only the booked time slots for a doctor on a date (no patient data), callable by the public booking page.

3. Important Notes
- A unique constraint on (doctor_id, appointment_date, slot_time) prevents two patients from booking the same slot.
- Seed data inserts four sample doctors only if the table is empty.
*/

CREATE TABLE IF NOT EXISTS doctors (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  name text NOT NULL,
  specialty text NOT NULL,
  qualification text NOT NULL,
  experience_years integer NOT NULL DEFAULT 0 CHECK (experience_years >= 0),
  bio text NOT NULL DEFAULT '',
  working_days integer[] NOT NULL DEFAULT '{1,2,3,4,5}',
  created_at timestamptz NOT NULL DEFAULT now()
);

CREATE TABLE IF NOT EXISTS appointments (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  doctor_id uuid NOT NULL REFERENCES doctors(id) ON DELETE RESTRICT,
  appointment_date date NOT NULL,
  slot_time time NOT NULL,
  patient_name text NOT NULL,
  patient_email text NOT NULL,
  patient_phone text NOT NULL,
  reason text NOT NULL DEFAULT '',
  status text NOT NULL DEFAULT 'confirmed' CHECK (status IN ('confirmed', 'cancelled')),
  created_at timestamptz NOT NULL DEFAULT now(),
  CONSTRAINT appointments_unique_slot UNIQUE (doctor_id, appointment_date, slot_time)
);

CREATE INDEX IF NOT EXISTS appointments_doctor_date_idx ON appointments (doctor_id, appointment_date);

ALTER TABLE doctors ENABLE ROW LEVEL SECURITY;
ALTER TABLE appointments ENABLE ROW LEVEL SECURITY;

GRANT SELECT ON doctors TO anon, authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE ON doctors TO service_role;
GRANT SELECT, INSERT, UPDATE, DELETE ON appointments TO service_role;

DROP POLICY IF EXISTS "public_select_doctors" ON doctors;
CREATE POLICY "public_select_doctors" ON doctors FOR SELECT
  TO anon, authenticated USING (true);

RETURNS SETOF time
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = public
AS $$
  SELECT a.slot_time FROM appointments a
  WHERE a.doctor_id = p_doctor_id
    AND a.appointment_date = p_date
    AND a.status = 'confirmed';
$$;
